\subsection{Semi-synthetic Data Based on BHP}
Controls $X$ from BHP data. 
$T | X \sim \mathcal{N}(\widehat{\mu}(X), \widehat{\sigma}^2(X))$, where $\widehat{\mu}$ and $\widehat{\sigma}^2$ are learnt from the real BHP data using RFs. $Y = f(T, X) + \epsilon$, $\epsilon \sim \mathcal{N}(0, 5.6 \mathrm{Var}(f(T,X)))$, where $5.6$ is the ratio of residual to explained variation after training a RF of $Y$ on $(T,X)$ in the real BHP data. 

We try 6 different $f$:
\begin{enumerate}
    \item Simple $f$: $f(T,X) = -0.6T$
    \item Simple $f$ with linear counfounders: $f(T,X) = -0.6T + X_{1:20}' b$, $b_j \sim \mathcal{U}[-0.5, 0.5]$ (drawn once per design, try 10 different random seeds).
    \item Simple $f$ with linear and non-linear counfounders:$f(T,X) = -0.6T + X_{1:20}' b + 1.5 \Lambda(10 * X_7) + 1.5\Lambda(10 * X_8)$ ($\Lambda$ is the expit function).
    \item Complex $f$: $f(T,X) = -(0.25 + X_1^2/20) T^3/3$
    \item Complex $f$ with linear confounders: $f(T,X) = -(0.25 + X_1^2/20 + X_{1:8}' c) T^3/3 + X_{1:20}' b$, $c_j \sim \mathcal{U}[-0.2, 0.2]$ (drawn once per design, try 10 different random seeds).
    \item Complex $f$ with linear and non-linear confounders: $f(T,X) = -(0.25 + X_1^2/20 + X_{1:8}' c) T^3/3 + X_{1:20}' b + 1.5 \Lambda(10 * X_7) + 1.5\Lambda(10 * X_8)$
\end{enumerate}
